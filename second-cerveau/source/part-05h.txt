
// V21 — Jesus Christ centre au-dessus de la grande ligne verticale, fleche sur la jonction

  try{
    document.title='Second Cerveau IA — V21';
    const pill=document.querySelector('.pill');
    if(pill)pill.textContent='Second Cerveau IA — V21';

    const versions=document.getElementById('versionList');
    if(versions && !versions.dataset.v21){
      versions.dataset.v21='1';
      versions.insertAdjacentHTML('afterbegin','<div class="item"><strong>V21 · actuelle</strong><small>Correction Bible Study Canvas : Jesus Christ est centre au-dessus du sommet de la grande ligne verticale et la fleche part de cette meme jonction.</small></div>');
      const old=[...versions.querySelectorAll('strong')].find(x=>x.textContent.includes('V20 · actuelle'));
      if(old)old.textContent='V20';
    }
  }catch(e){}

  const section=document.getElementById('page-bible-canvas');
  if(!section)return;
  const right=section.querySelector('.bsc-right');
  if(!right)return;

  const style=document.createElement('style');
  style.textContent=`

    }

      content:''!important;

    }

    #page-bible-canvas .bsc-verse{margin-top:145px!important;}

  document.head.appendChild(style);

  let marker=right.querySelector('.bsc-top-marker');
  if(!marker){
    marker=document.createElement('div');
    marker.className='bsc-top-marker';
    right.insertBefore(marker,right.firstChild);
  }
  marker.innerHTML='';

  const label=document.createElement('span');
  label.className='bsc-symbol';
  label.textContent='Jesus Christ';
  marker.appendChild(label);

  const junction=document.createElement('span');
  junction.className='bsc-divider-junction';
  junction.setAttribute('aria-hidden','true');
  marker.appendChild(junction);

  const arrow=document.createElement('span');
  arrow.className='bsc-arrow-line';
  arrow.setAttribute('aria-hidden','true');
  marker.appendChild(arrow);
})();
