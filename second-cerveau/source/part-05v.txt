    const versions=document.getElementById('versionList');
    if(versions&&!versions.dataset.v35){
      versions.dataset.v35='1';
      versions.insertAdjacentHTML('afterbegin','<div class="item"><strong>V35 · actuelle</strong><small>Les 20 méthodes peuvent maintenant être utilisées dans des fenêtres interactives : choix du certificat IBM, sujet, exercices adaptés, réponses et autosauvegarde.</small></div>');
      const old=[...versions.querySelectorAll('strong')].find(x=>x.textContent.includes('V34 · actuelle'));if(old)old.textContent='V34';
    }
  }catch(e){}

  if(!guide)return;

  const style=document.createElement('style');
  style.textContent=`
    .use-method-v35{margin-top:3px}

          </select>
        </div>
        <div class="fieldgrp" style="margin:0">
          <label class="label">Sujet à travailler</label>
          <input class="field" id="methodTopicV35" placeholder="Ex. Python — fonctions, Docker, REST API...">
        </div>
      </div>
      <div class="method-fields-v35" id="methodFieldsV35"></div>
      <div class="actions" style="margin-top:12px">
        <button class="btn primary" id="methodSaveV35" type="button">Enregistrer</button>
        <button class="btn" id="methodClearV35" type="button">Effacer cette méthode</button>
        <button class="btn" id="methodCloseBottomV35" type="button">Fermer</button>
      </div>
      <div class="method-status-v35" id="methodStatusV35">Les réponses sont autosauvegardées sur cet appareil.</div>
    </div>
  `;
  document.body.appendChild(modal);

  let activeIndex=0;
  let saveTimer=null;

  function readAll(){
    try{
      const v=JSON.parse(localStorage.getItem(WORK_KEY)||'{}');
      return v&&typeof v==='object'?v:{};
    }catch(e){return{}}
  }

  function writeAll(all,message='Autosauvegardé.'){
    try{
      localStorage.setItem(WORK_KEY,JSON.stringify(all));
      document.getElementById('methodStatusV35').textContent=message;
      return true;
    }catch(e){
      document.getElementById('methodStatusV35').textContent='Impossible de sauvegarder : le stockage du navigateur est plein.';
      return false;
    }
  }

  function collect(){
    return{
      certificate:document.getElementById('methodCertV35').value,
      topic:document.getElementById('methodTopicV35').value,
      answers:[...document.querySelectorAll('#methodFieldsV35 textarea')].map(x=>x.value),
      updatedAt:new Date().toISOString()
    };
  }

  function saveCurrent(message='Autosauvegardé.'){
    const all=readAll();
    all[String(activeIndex)]=collect();
    writeAll(all,message);
  }

  function queueSave(){
    clearTimeout(saveTimer);
    saveTimer=setTimeout(()=>saveCurrent(),250);
  }

  function openMethod(index){
    activeIndex=index;
    const method=METHODS[index];
    const all=readAll();
    const saved=all[String(index)]||{};
    const globalTopic=document.getElementById('learnTopic')?.value||'';

    document.getElementById('methodTitleV35').textContent=(index+1)+'. '+method.title;
    document.getElementById('methodSubtitleV35').textContent='Utilise cette méthode directement pour travailler une notion de tes certificats IBM.';
    document.getElementById('methodCertV35').value=saved.certificate||'IBM Full Stack Software Developer';
    document.getElementById('methodTopicV35').value=saved.topic||globalTopic;

    const fields=document.getElementById('methodFieldsV35');
    fields.innerHTML=method.fields.map((label,i)=>`
      <div class="method-field-v35">
        <label>${label}</label>
        <textarea class="field" data-method-answer="${i}" placeholder="Écris ta réponse ici..."></textarea>
      </div>
    `).join('');

    [...fields.querySelectorAll('textarea')].forEach((el,i)=>{
      el.value=Array.isArray(saved.answers)?(saved.answers[i]||''):'';
      el.addEventListener('input',queueSave);
    });
    document.getElementById('methodCertV35').onchange=queueSave;
    document.getElementById('methodTopicV35').oninput=queueSave;

    const status=document.getElementById('methodStatusV35');
    status.textContent=saved.updatedAt?'Dernière sauvegarde : '+new Date(saved.updatedAt).toLocaleString('fr-CA'):'Les réponses sont autosauvegardées sur cet appareil.';
    modal.classList.add('open');
    setTimeout(()=>document.getElementById('methodTopicV35').focus(),40);
  }

  function closeMethod(){
    clearTimeout(saveTimer);
    if(modal.classList.contains('open'))saveCurrent('Sauvegardé.');
    modal.classList.remove('open');
  }

  const details=[...guide.querySelectorAll('.method-v34')];
  details.forEach((detail,index)=>{
    const body=detail.querySelector('.method-v34-body');
    if(!body||body.querySelector('.use-method-v35'))return;
    const row=document.createElement('div');
    row.className='actions use-method-v35';
    row.innerHTML='<button class="btn primary" type="button">Utiliser cette méthode</button>';
    row.querySelector('button').onclick=()=>openMethod(index);
    body.appendChild(row);
  });

  document.getElementById('methodSaveV35').onclick=()=>saveCurrent('Méthode enregistrée.');
  document.getElementById('methodClearV35').onclick=()=>{
    if(!confirm('Effacer les réponses enregistrées pour cette méthode ?'))return;
    const all=readAll();
    delete all[String(activeIndex)];
    if(writeAll(all,'Réponses effacées.')){
      document.getElementById('methodTopicV35').value=document.getElementById('learnTopic')?.value||'';
      document.querySelectorAll('#methodFieldsV35 textarea').forEach(x=>x.value='');
    }
  };
  document.getElementById('methodCloseV35').onclick=closeMethod;
  document.getElementById('methodCloseBottomV35').onclick=closeMethod;
  modal.addEventListener('click',e=>{if(e.target===modal)closeMethod()});
  document.addEventListener('keydown',e=>{if(e.key==='Escape'&&modal.classList.contains('open'))closeMethod()});
})();
