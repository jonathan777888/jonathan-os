  function collectReport(){
    syncAll();
    const learn=readJson(LEARN_KEY,{})||{};
    const work=readJson(WORK_KEY,{})||{};
    const c=readJson(CORNELL_KEY,{})||{};
    let answered=0,total=0;

    const core=coreQuestions.map(([label,key])=>{
      total++;const value=clean(learn[key]);if(value)answered++;
      return{label,value:answer(value)};
    });

    const methods=METHOD_DEFS.map(([title,fields],index)=>{
      const saved=work[String(index)]||{};
      const answers=Array.isArray(saved.answers)?saved.answers:[];
      const items=fields.map((label,i)=>{
        total++;const value=clean(answers[i]);if(value)answered++;
        return{label,value:answer(value)};
      });
      return{title,topic:answer(saved.topic),items};
    });

    const cornellItems=[
      ['Titre','title'],['Mots-clés','keywords'],['Formules','formulas'],['Doutes','doubts'],
      ['Notions essentielles','essentials'],['Dates','dates'],['Auteurs','authors'],['Prise de notes','notes'],['Synthèse','summary']
    ].map(([label,key])=>{
      total++;const value=clean(c[key]);if(value)answered++;
      return{label,key,value:answer(value)};
    });

    return{
      generatedAt:new Date().toISOString(),
      topic:answer(learn.topic),
      core,
      review:[['J+1',!!learn.reviewJ1],['J+3',!!learn.reviewJ3],['J+7',!!learn.reviewJ7],['J+14',!!learn.reviewJ14],['J+30',!!learn.reviewJ30]],
      methods,cornell:cornellItems,answered,total,missing:total-answered
    };
  }

  function esc(s){return String(s??'').replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]))}
