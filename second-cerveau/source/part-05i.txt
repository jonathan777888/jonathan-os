  function collectCorpus(){

    }

    }

  }
  function tokens(s){return [...new Set(norm(s).split(' ').filter(x=>x.length>2))]}
  function retrieve(question){
    const qt=tokens(question),corpus=collectCorpus();
    const scored=corpus.map(x=>{const title=norm(x.title),hay=norm(x.title+' '+x.text);let score=0;qt.forEach(t=>{if(title.includes(t))score+=5;if(hay.includes(t))score+=2});if(x.kind==='Tâche'&&/open|ouvert/i.test(x.text))score+=2;return{...x,score}});
    let chosen=scored.filter(x=>x.score>0).sort((a,b)=>b.score-a.score).slice(0,mobile?7:12);
    const open=scored.filter(x=>x.kind==='Tâche'&&/open|ouvert/i.test(x.text)).slice(0,4);
    open.forEach(x=>{if(!chosen.some(y=>y.kind===x.kind&&y.ref===x.ref))chosen.push(x)});
    if(!chosen.length)chosen=scored.slice(0,mobile?6:10);
    return chosen.slice(0,mobile?9:14);
  }
  function buildContext(items){return items.map((x,i)=>`${i+1}. [${x.kind}] ${x.title}\n${x.text}`).join('\n\n').slice(0,mobile?4300:11000)}
  function fallback(question,items){

    text+='\nUtilise ces éléments comme base de décision. Je n’ajoute pas d’information personnelle qui ne soit pas déjà présente dans ton Second Cerveau.';
    return text;
  }
  function reliable(text){if(!text||text.trim().length<35)return false;const words=norm(text).split(' ').filter(Boolean);if(words.length<10)return false;return new Set(words).size/words.length>.35}

    import { pipeline, env } from "https://cdn.jsdelivr.net/npm/@huggingface/transformers@4.0.1";
    env.allowLocalModels=false;
    let generator=null;
    const mobile=/Android|iPhone|iPad|iPod|Mobile/i.test(navigator.userAgent)||(navigator.deviceMemory&&navigator.deviceMemory<=4);

      const options=mobile?{dtype:"q4"}:(navigator.gpu?{device:"webgpu",dtype:"q4f16"}:{dtype:"q4"});

        self.postMessage({type:"generating",mobile});
        const out=await pipe(messages,{max_new_tokens:mobile?100:230,do_sample:false,repetition_penalty:1.18,no_repeat_ngram_size:4});
        let generated=out?.[0]?.generated_text;let answer=Array.isArray(generated)?generated.at(-1)?.content||"":String(generated||"");
        answer=answer.replace(/<think>[\s\S]*?<\/think>/gi,"").replace(/^assistant\s*/i,"").trim();
        self.postMessage({type:"answer",answer,mobile});
      }catch(err){self.postMessage({type:"error",message:err?.message||"Le modèle local n'a pas pu démarrer.",mobile});}
    };
  `;

  function getWorker(){
    if(worker)return worker;
    const url=URL.createObjectURL(new Blob([workerCode],{type:'text/javascript'}));
    worker=new Worker(url,{type:'module'});URL.revokeObjectURL(url);
    worker.onmessage=e=>{
      const d=e.data||{};
      if(d.type==='loading')statusEl.textContent=d.mobile?'Préparation du modèle léger sur ton appareil…':'Préparation du modèle IA local…';
      else if(d.type==='progress'){const p=d.progress||{},pct=Number.isFinite(p.progress)?' '+Math.round(p.progress)+' %':'';statusEl.textContent='Téléchargement du modèle'+pct+' — garde cette page ouverte.'}

      else if(d.type==='generating')statusEl.textContent='L’agent réfléchit à partir de ton contexte…';

      else if(d.type==='error'){clearTimeout(responseTimer);askBtn.disabled=false;lastAnswer=fallback(lastQuestion,lastSources);answerEl.textContent=lastAnswer;statusEl.textContent='Le moteur local n’a pas pu démarrer.   }
