(()=>{

  const WORK_KEY='second-brain-method-workspaces-v35';
  const METHOD_DEFS=[["Méthode Feynman",["Explique la notion comme à un débutant","Quels mots ou passages sont encore difficiles ?","Réexplique encore plus simplement","Quelle question poserais-tu pour vérifier que l’autre personne a compris ?"]],["Rappel actif",["Écris les questions auxquelles tu veux répondre sans regarder le cours","Réponds de mémoire","Vérifie ensuite : qu’est-ce qui était correct ?","Qu’est-ce que tu dois corriger ou revoir ?"]],["Répétition espacée",["Qu’est-ce que tu arrives encore à rappeler sans aide ?","Qu’as-tu oublié depuis la dernière révision ?","Réponse corrigée / notion à renforcer","Quand feras-tu la prochaine révision ?"]],["Exercices pratiques",["Quel exercice ou petit programme dois-tu réaliser ?","Écris ton raisonnement ou ton code","Quel résultat as-tu obtenu ?","Que vas-tu modifier pour le prochain essai ?"]],["Pourquoi / Comment",["Quelle ligne, commande ou notion veux-tu comprendre ?","Pourquoi existe-t-elle ?","Comment l’ordinateur l’exécute-t-il ?","Que se passe-t-il si tu la retires ou la modifies ?"]],["Carte mentale / conceptuelle",["Quel est le concept central ?","Quels sont les composants principaux ?","Comment ces composants sont-ils reliés ?","Résume le flux complet avec des flèches ou des relations"]],["Double codage",["Explique la notion avec des mots","Dessine-la avec du texte, des flèches ou un mini-schéma","Quel élément visuel correspond à quelle partie du code ?","Que comprends-tu mieux grâce au schéma ?"]],["Exemples concrets",["Quelle notion abstraite veux-tu comprendre ?","Donne un exemple dans une situation réelle","Transforme cet exemple en petit exemple de code","Pourquoi cet exemple représente-t-il bien la notion ?"]],["Comparaison",["Concept / technologie A","Concept / technologie B","Ressemblances et différences","Quand utiliser A ? ?","Quelle est la prochaine petite étape de code ?"]],["Entrelacement",["Quels sujets vas-tu mélanger aujourd’hui ?","Exercice 1 et ta réponse","Exercice 2 et ta réponse","Exercice 3 : quelle méthode ou technologie fallait-il reconnaître ?"]],["Auto-évaluation",["Écris tes questions de test","Réponds sans regarder les notes","Résultat : maîtrisé / hésitation / erreur","Quelles notions dois-tu   try{
    document.title='Second Cerveau IA — V36';
    const pill=document.querySelector('.pill');if(pill)pill.textContent='Second Cerveau IA — V36';
    const vs=document.getElementById('versionStatus');if(vs)vs.textContent='Version actuelle V36 · cette version hérite directement de V35 et ajoute l’export Word (.docx) de toutes les réponses d’apprentissage.';
    const versions=document.getElementById('versionList');
    if(versions&&!versions.dataset.v36){
      versions.dataset.v36='1';
      versions.insertAdjacentHTML('afterbegin','<div class="item"><strong>V36 · actuelle</strong><small>Export Word (.docx) : toutes les questions et réponses de l’onglet Apprentissage et des 20 méthodes sont exportées. Les réponses manquantes sont marquées « NON RÉPONDU » pour faciliter la vérification.</small></div>');
      const old=[...versions.querySelectorAll('strong')].find(x=>x.textContent.includes('V35 · actuelle'));if(old)old.textContent='V35';
    }
  }catch(e){}

  const page=document.getElementById('page-learning');
  if(!page)return;

  function readJson(key,fallback){
    try{return JSON.parse(localStorage.getItem(key)||'null')??fallback}catch(e){return fallback}
  }
  const clean=v=>String(v??'').trim();
  const answer=v=>clean(v)||'NON RÉPONDU';

  const coreQuestions=[
    ['What? — C’est quoi ?','what'],
    ['Who? — Qui est concerné ?','who'],
    ['Where? — Où cela se produit / s’applique ?','where'],
    ['When? — Quand cela intervient ?','when'],
    ['Why? — Pourquoi ?','why'],
    ['How? — Comment ça fonctionne ?','how'],
    ['What if? — Que se passe-t-il si… ?','whatif'],
    ['Méthode Feynman — Explique avec tes propres mots','feynman'],
    ['Exemple concret','example'],
    ['Analogie','analogy'],
    ['Exercices pratiques / rappel actif','practice'],
    ['Méthode des erreurs','errors'],
    ['Mini-examen','miniTest']
  ];

  function syncOpenMethod(){
    const modal=document.getElementById('methodModalV35');
    if(modal&&modal.classList.contains('open')){
      const btn=document.getElementById('methodSaveV35');
      if(btn)btn.click();
    }
    const btn=document.getElementById('learningSave');
    if(btn)btn.click();
  }

  function collectReport(){
    syncOpenMethod();
    const learn=readJson(LEARN_KEY,{})||{};
    const work=readJson(WORK_KEY,{})||{};
    let answered=0,total=0;

    const core=coreQuestions.map(([label,key])=>{
      total++;
      const value=clean(learn[key]);
      if(value)answered++;
      return {label,value:answer(value)};
    });

    const methods=METHOD_DEFS.map(([title,fields],index)=>{
      const saved=work[String(index)]||{};
      const answers=Array.isArray(saved.answers)?saved.answers:[];
      const items=fields.map((label,i)=>{
        total++;
        const value=clean(answers[i]);
        if(value)answered++;
        return {label,value:answer(value)};
      });
      return{
        title,
        certificate:answer(saved.certificate),
        topic:answer(saved.topic),
        updatedAt:saved.updatedAt||'',
        items
      };
    });

    return{
      generatedAt:new Date().toISOString(),
      topic:answer(learn.topic),
      core,
      review:[
        ['J+1',!!learn.reviewJ1],['J+3',!!learn.reviewJ3],['J+7',!!learn.reviewJ7],['J+14',!!learn.reviewJ14],['J+30',!!learn.reviewJ30]
      ],
      methods,
      answered,total,
      missing:total-answered
    };
  }

  async function exportDocx(){
    const status=document.getElementById('learningExportStatusV36');
    if(status)status.textContent='Préparation du document Word…';
    const report=collectReport();

    try{
      const docx=await import('https://cdn.jsdelivr.net/npm/docx@9.7.2/+esm');
      const {Document,Packer,Paragraph,TextRun,HeadingLevel}=docx;
      const children=[];

      children.push(new Paragraph({text:'Second Cerveau — Rapport d’apprentissage',heading:HeadingLevel.TITLE}));
      children.push(new Paragraph({children:[new TextRun({text:'Sujet principal : ',bold:true}),new TextRun(report.topic)]}));
      children.push(new Paragraph({children:[new TextRun({text:'Date d’export : ',bold:true}),new TextRun(new Date(report.generatedAt).toLocaleString('fr-CA'))]}));
      children.push(new Paragraph({children:[
        new TextRun({text:'Progression : ',bold:true}),
        new TextRun(report.answered+' réponses sur '+report.total+' questions · '+report.missing+' non répondues')
      ]}));
      children.push(new Paragraph({children:[new TextRun({text:'Objectif : ',bold:true}),new TextRun('Envoyer ce fichier à ChatGPT pour vérifier si chaque question a été répondue correctement et complètement.')]}));

      children.push(new Paragraph({text:'Partie 1 — Cycle principal d’apprentissage',heading:HeadingLevel.HEADING_1}));
      report.core.forEach((q,i)=>{
        children.push(new Paragraph({children:[new TextRun({text:(i+1)+'. '+q.label,bold:true})]}));
        children.push(new Paragraph({text:q.value}));
      });

      children.push(new Paragraph({text:'Répétition espacée',heading:HeadingLevel.HEADING_2}));
      report.review.forEach(([label,done])=>{
        children.push(new Paragraph({text:label+' : '+(done?'FAIT':'NON FAIT')}));
      });

      children.push(new Paragraph({text:'Partie 2 — Les 20 méthodes interactives',heading:HeadingLevel.HEADING_1}));
      report.methods.forEach((m,mi)=>{
        children.push(new Paragraph({text:(mi+1)+'. '+m.title,heading:HeadingLevel.HEADING_2}));
        children.push(new Paragraph({children:[new TextRun({text:'Certificat / domaine : ',bold:true}),new TextRun(m.certificate)]}));
        children.push(new Paragraph({children:[new TextRun({text:'Sujet : ',bold:true}),new TextRun(m.topic)]}));
        m.items.forEach((q,qi)=>{
          children.push(new Paragraph({children:[new TextRun({text:(qi+1)+'. '+q.label,bold:true})]}));
          children.push(new Paragraph({text:q.value}));
        });
      });

      children.push(new Paragraph({text:'Résumé des éléments manquants',heading:HeadingLevel.HEADING_1}));
      if(report.missing===0){
        children.push(new Paragraph({text:'Toutes les questions ont une réponse.'}));
      }else{
        children.push(new Paragraph({text:report.missing+' question(s) sont encore marquées NON RÉPONDU.'}));
      }

      const doc=new Document({sections:[{properties:{},children}]});
      const blob=await Packer.toBlob(doc);
      const url=URL.createObjectURL(blob);
      const a=document.createElement('a');
      const safeTopic=clean(report.topic).replace(/[^a-zA-Z0-9À-ÿ_-]+/g,'-').replace(/^-+|-+$/g,'').slice(0,50)||'apprentissage';
      a.href=url;
      a.download='Second-Cerveau-'+safeTopic+'-'+new Date().toISOString().slice(0,10)+'.docx';
      document.body.appendChild(a);
      a.click();
      a.remove();
      setTimeout(()=>URL.revokeObjectURL(url),1500);
      if(status)status.textContent='Document Word exporté : '+report.answered+'/'+report.total+' questions répondues.';
    }catch(err){
      console.error('V36 DOCX export failed',err);
      exportWordFallback(report);
      if(status)status.textContent='Le format .docx n’a pas pu être chargé. Une version Word compatible (.doc) a été exportée à la place.';
    }
  }

  function esc(s){
    return String(s??'').replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]));
  }

  function exportWordFallback(report){
    let html='<!doctype html><html><head><meta charset="utf-8"><title>Rapport apprentissage</title><style>body{font-family:Arial,sans-serif;line-height:1.45}h1,h2,h3{margin-top:24px}.q{font-weight:bold;margin-top:14px}.a{white-space:pre-wrap;margin:5px 0 12px}.missing{font-weight:bold;color:#a00}</style></head><body>';
    html+='<h1>Second Cerveau — Rapport d’apprentissage</h1>';
    html+='<p><b>Sujet principal :</b> '+esc(report.topic)+'</p>';
    html+='<p><b>Progression :</b> '+report.answered+' / '+report.total+' questions répondues · '+report.missing+' non répondues</p>';
    html+='<h2>Partie 1 — Cycle principal d’apprentissage</h2>';
    report.core.forEach((q,i)=>{html+='<div class="q">'+(i+1)+'. '+esc(q.label)+'</div><div class="a'+(q.value==='NON RÉPONDU'?' missing':'')+'">'+esc(q.value)+'</div>'});
    html+='<h3>Répétition espacée</h3>';
    report.review.forEach(([label,done])=>{html+='<p>'+esc(label)+' : '+(done?'FAIT':'NON FAIT')+'</p>'});
    html+='<h2>Partie 2 — Les 20 méthodes interactives</h2>';
    report.methods.forEach((m,mi)=>{
      html+='<h3>'+(mi+1)+'. '+esc(m.title)+'</h3><p><b>Certificat :</b> '+esc(m.certificate)+'</p><p><b>Sujet :</b> '+esc(m.topic)+'</p>';
      m.items.forEach((q,qi)=>{html+='<div class="q">'+(qi+1)+'. '+esc(q.label)+'</div><div class="a'+(q.value==='NON RÉPONDU'?' missing':'')+'">'+esc(q.value)+'</div>'});
    });
    html+='</body></html>';
    const blob=new Blob([html],{type:'application/msword;charset=utf-8'});
    const url=URL.createObjectURL(blob);
    const a=document.createElement('a');
    a.href=url;a.download='Second-Cerveau-apprentissage-'+new Date().toISOString().slice(0,10)+'.doc';
    document.body.appendChild(a);a.click();a.remove();setTimeout(()=>URL.revokeObjectURL(url),1500);
  }

  function addExportButtons(){
    const topRow=page.querySelector('.section-card > .row');
    const topActions=topRow?.querySelector('.actions');
    if(topActions&&!document.getElementById('learningExportWordV36')){
      const btn=document.createElement('button');
      btn.className='btn';
      btn.id='learningExportWordV36';
      btn.type='button';
      btn.textContent='Exporter Word (.docx)';
      btn.onclick=exportDocx;
      topActions.appendChild(btn);

      const status=document.createElement('div');
      status.className='status';
      status.id='learningExportStatusV36';
      status.style.width='100%';
      status.textContent='Le Word inclura toutes les questions, réponses et les éléments NON RÉPONDU.';
      topRow.insertAdjacentElement('afterend',status);
    }

    const modalActions=document.querySelector('#methodModalV35 .actions');
    if(modalActions&&!document.getElementById('methodExportWordV36')){
      const btn=document.createElement('button');
      btn.className='btn';
      btn.id='methodExportWordV36';
      btn.type='button';
      btn.textContent='Exporter tout en Word';
      btn.onclick=exportDocx;
      modalActions.insertBefore(btn,modalActions.lastElementChild);
    }
  }

  addExportButtons();
})();
