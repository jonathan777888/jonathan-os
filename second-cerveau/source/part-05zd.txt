  function collectAnswers(){
    const learn=readJson(LEARN_KEY,{})||{};
    const work=readJson(WORK_KEY,{})||{};
    const out=[];
    coreDefs.forEach(([key,q],i)=>{
      const a=clean(learn[key]);
      if(a) out.push({id:'core-'+key,question:q,answer:a});
    });
    methodNames.forEach((name,mi)=>{
      const saved=work[String(mi)]||{};
      const answers=Array.isArray(saved.answers)?saved.answers:[];
      answers.forEach((a,qi)=>{
        const value=clean(a);
        if(value) out.push({id:'m'+(mi+1)+'-q'+(qi+1),question:name+' — '+(methodPrompts[mi]?.[qi]||('Question '+(qi+1))),answer:value});
      });
    });
    return out;
  }

  function escapeHtml(s){return String(s??'').replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]))}

  async function extractDocx(file){
    const mod=await import('https://cdn.jsdelivr.net/npm/mammoth@1.9.0/+esm');
    const buf=await file.arrayBuffer();
    const res=await mod.extractRawText({arrayBuffer:buf});
    return res.value||'';
  }

  async function extractPdf(file){
    const pdfjs=await import('https://cdn.jsdelivr.net/npm/pdfjs-dist@4.10.38/build/pdf.min.mjs');
    pdfjs.GlobalWorkerOptions.workerSrc='https://cdn.jsdelivr.net/npm/pdfjs-dist@4.10.38/build/pdf.worker.min.mjs';
    const buf=await file.arrayBuffer();
    const pdf=await pdfjs.getDocument({data:buf}).promise;
    let text='';
    for(let i=1;i<=pdf.numPages;i++){
      const page=await pdf.getPage(i);
      const content=await page.getTextContent();
      text+='\n'+content.items.map(x=>x.str||'').join(' ');
    }
    return text;
  }

  async function extractFile(file){
    const n=file.name.toLowerCase();
    if(n.endsWith('.docx')) return await extractDocx(file);
    if(n.endsWith('.pdf')) return await extractPdf(file);
    return await file.text();
  }

  function renderDocs(){
    docsEl.innerHTML=documents.length?documents.map((d,i)=>`
      <div class="verify-doc-v43"><strong>${escapeHtml(d.name)}</strong><br><span class="muted">${Math.round(d.text.length/100)/10} k caractères</span>
      <button class="btn" data-remove-doc="${i}" style="float:right;min-height:26px;padding:2px 8px">×</button></div>
    `).join(''):'<div class="muted">Aucun document ajouté.</div>';
    docsEl.querySelectorAll('[data-remove-doc]').forEach(b=>b.onclick=()=>{documents.splice(Number(b.dataset.removeDoc),1);renderDocs()});
  }

    for(const file of files){
      statusEl.textContent='Lecture de '+file.name+'…';
      try{
        const text=clean(await extractFile(file)).slice(0,50000);
        if(text) documents.push({name:file.name,text});
      }catch(e){
        statusEl.textContent='Impossible de lire '+file.name+'.';
      }
    }
    filesEl.value='';
    renderDocs();
    statusEl.textContent=documents.length+' document(s) prêt(s).';
  };

  function applyCornell(c){
    if(!c)return;
    const keywords=document.getElementById('cornellKeywordsV37');
    const notes=document.getElementById('cornellNotesV37');
    const summary=document.getElementById('cornellSummaryV37');
    if(keywords) keywords.value=(c.keywords||[]).join(' · ');
    if(notes) notes.value=c.notes||'';
    if(summary) summary.value=c.summary||'';
    const old=readJson(CORNELL_KEY,{})||{};
    try{
      localStorage.setItem(CORNELL_KEY,JSON.stringify({...old,keywords:(c.keywords||[]).join(' · '),notes:c.notes||'',summary:c.summary||'',verifiedByV43:true,updatedAt:new Date().toISOString()}));
    }catch(e){}
  }

  function renderResults(data,answers){
    const map=new Map(answers.map(a=>[a.id,a]));
    const sourceTitle=new Map((data.sources||[]).map(s=>[s.url,s.title||s.url]));
    const items=data.items||[];
    resultsEl.innerHTML=`
      <div class="verify-summary-v43"><strong>Vérification terminée</strong><br>
      <span class="muted">${items.length} réponse(s) vérifiée(s) · ${(data.sources||[]).length} source(s) web trouvée(s) · ${documents.length} document(s) fourni(s).</span></div>
      ${items.map(item=>{
        const original=map.get(item.id)||{};
        return `<div class="verify-result-v43">
          <span class="verify-status-v43">${escapeHtml(statusLabels[item.status]||item.status)}</span>
          <div><strong>Question</strong><div class="verify-answer-v43">${escapeHtml(original.question||'')}</div></div>
          <div style="margin-top:8px"><strong>Ta réponse originale</strong><div class="verify-answer-v43">${escapeHtml(original.answer||'')}</div></div>
          <div style="margin-top:8px"><strong>Réponse vérifiée et corrigée</strong><div class="verify-answer-v43">${escapeHtml(item.corrected_answer||'')}</div></div>
          <div style="margin-top:8px"><strong>Pourquoi</strong><div class="verify-answer-v43">${escapeHtml(item.explanation||'')}</div></div>
          ${(item.document_support||[]).length?'<div style="margin-top:8px"><strong>Documents utilisés</strong><div>'+item.document_support.map(x=>escapeHtml(x)).join(' · ')+'</div></div>':''}
          ${(item.source_urls||[]).length?'<div class="verify-sources-v43" style="margin-top:8px"><strong>Sources web</strong>'+item.source_urls.map(u=>'<a href="'+escapeHtml(u)+'" target="_blank" rel="noopener">'+escapeHtml(sourceTitle.get(u)||u)+'</a>').join('')+'</div>':''}
        </div>`;
      }).join('')}
    `;
  }

      });

        if(data.error==='setup_required') throw new Error('Le serveur V43 n’a pas encore sa clé IA privée configurée.');
        if(data.error==='unauthorized') throw new Error('Code d’accès V43 incorrect.');
        throw new Error('Erreur de vérification : '+(data.error||res.status));
