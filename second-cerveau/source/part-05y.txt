
// V38 — Cornell automatique à partir des exercices et méthodes
(()=>{
  const CORNELL_KEY='second-brain-cornell-v37';
  const LEARN_KEY='second-brain-learning-v33';
  const WORK_KEY='second-brain-method-workspaces-v35';

  const METHOD_DEFS=[
    ['Méthode Feynman',['Explique la notion comme à un débutant','Quels mots ou passages sont encore difficiles ?','Réexplique encore plus simplement','Quelle question poserais-tu pour vérifier que l’autre personne a compris ?']],
    ['Rappel actif',['Écris les questions auxquelles tu veux répondre sans regarder le cours','Réponds de mémoire','Vérifie ensuite : qu’est-ce qui était correct ?','Qu’est-ce que tu dois corriger ou revoir ?']],
    ['Répétition espacée',['Qu’est-ce que tu arrives encore à rappeler sans aide ?','Qu’as-tu oublié depuis la dernière révision ?','Réponse corrigée / notion à renforcer','Quand feras-tu la prochaine révision ?']],
    ['Exercices pratiques',['Quel exercice ou petit programme dois-tu réaliser ?','Écris ton raisonnement ou ton code','Quel résultat as-tu obtenu ?','Que vas-tu modifier pour le prochain essai ?']],
    ['Pourquoi / Comment',['Quelle ligne, commande ou notion veux-tu comprendre ?','Pourquoi existe-t-elle ?','Comment l’ordinateur l’exécute-t-il ?','Que se passe-t-il si tu la retires ou la modifies ?']],
    ['Carte mentale / conceptuelle',['Quel est le concept central ?','Quels sont les composants principaux ?','Comment ces composants sont-ils reliés ?','Résume le flux complet avec des flèches ou des relations']],
    ['Double codage',['Explique la notion avec des mots','Dessine-la avec du texte, des flèches ou un mini-schéma','Quel élément visuel correspond à quelle partie du code ?','Que comprends-tu mieux grâce au schéma ?']],
    ['Exemples concrets',['Quelle notion abstraite veux-tu comprendre ?','Donne un exemple dans une situation réelle','Transforme cet exemple en petit exemple de code','Pourquoi cet exemple représente-t-il bien la notion ?']],
    ['Comparaison',['Concept / technologie A','Concept / technologie B','Ressemblances et différences','Quand utiliser A ? Quand utiliser B ?']],
    ['Apprentissage par projet',['Quel produit ou mini-projet veux-tu construire ?','Quelles fonctionnalités faut-il créer ?','Quelles compétences sont nécessaires ?','Quelle est la prochaine petite étape ?']],
    ['Entrelacement',['Quels sujets vas-tu mélanger aujourd’hui ?','Exercice 1 et ta réponse','Exercice 2 et ta réponse','Exercice 3 : quelle méthode ou technologie fallait-il reconnaître ?']],
    ['Auto-évaluation',['Écris tes questions de test','Réponds sans regarder les notes','Résultat : maîtrisé / hésitation / erreur','Quelles notions dois-tu retravailler ?']],
    ['Méthode Cornell',['Notes principales','Colonne questions / mots-clés','Résumé du sujet en quelques phrases','Réponds maintenant aux questions sans regarder tes notes']],
    ['SQ3R',['Survey : quels titres et sections vois-tu ?','Question : transforme les titres en questions','Read + Recite : réponds avec tes propres mots','Review : quels points dois-tu revoir ?']],
    ['Enseigner à quelqu’un',['Prépare ta mini-leçon','Donne un exemple concret','Quelle erreur fréquente veux-tu expliquer ?','Pose une question finale et donne la réponse']],
    ['Méthode des erreurs',['Erreur / mauvaise réponse','Quelle était la cause réelle ?','Quelle correction as-tu faite ?','Règle à retenir + nouvel exercice similaire']],
    ['Analogies',['Quelle nouvelle notion veux-tu comprendre ?','À quoi de connu peux-tu la comparer ?','Quelles parties correspondent entre les deux ?','Où l’analogie devient-elle incorrecte ou limitée ?']],
    ['Chunking',['Quel gros sujet veux-tu découper ?','Bloc 1 et son rôle','Blocs suivants et leurs rôles','Comment tous les blocs fonctionnent-ils ensemble ?']],
    ['Pré-test',['Écris 5 questions avant d’étudier','Tes réponses initiales, même incertaines','Après le cours : quelles réponses changent ?','Qu’as-tu appris que tu ne savais pas au départ ?']],
    ['Révision cumulative',['Quelle nouvelle notion étudies-tu aujourd’hui ?','Quelles anciennes notions vas-tu revoir ?','Réponds à quelques questions mélangées','Quelles anciennes notions commencent à s’affaiblir ?']]
  ];

  try{
    document.title='Second Cerveau IA — V38';
    const pill=document.querySelector('.pill');if(pill)pill.textContent='Second Cerveau IA — V38';
    const vs=document.getElementById('versionStatus');if(vs)vs.textContent='Version actuelle V38 · cette version hérite directement de V37 et relie automatiquement la fiche Cornell aux exercices et aux 20 méthodes.';
    const versions=document.getElementById('versionList');
    if(versions&&!versions.dataset.v38){
      versions.dataset.v38='1';
      versions.insertAdjacentHTML('afterbegin','<div class="item"><strong>V38 · actuelle</strong><small>La fiche Cornell se remplit automatiquement depuis les exercices : titre, prise de notes, mots-clés et synthèse. Les autres zones restent manuelles.</small></div>');
      const old=[...versions.querySelectorAll('strong')].find(x=>x.textContent.includes('V37 · actuelle'));if(old)old.textContent='V37';
    }
  }catch(e){}

  const page=document.getElementById('page-learning');
  const sheet=document.getElementById('cornellSheetV37');
  if(!page||!sheet)return;

  const readJson=(key,fallback)=>{try{return JSON.parse(localStorage.getItem(key)||'null')??fallback}catch(e){return fallback}};
  const clean=v=>String(v??'').trim();

  const labels={
    title:document.querySelector('.cornell-v37-title input'),
    keywords:document.getElementById('cornellKeywordsV37'),
    notes:document.getElementById('cornellNotesV37'),
    summary:document.getElementById('cornellSummaryV37')
  };

  if(labels.title){
    labels.title.readOnly=true;
    labels.title.title='Rempli automatiquement à partir du sujet principal';
  }
  [labels.keywords,labels.notes,labels.summary].forEach(el=>{
    if(!el)return;
    el.readOnly=true;
    el.title='Rempli automatiquement à partir de tes réponses';
    el.style.background='#f7faf5';
  });

  const keywordLabel=labels.keywords?.previousElementSibling;
  if(keywordLabel)keywordLabel.textContent='Mots-clés — automatique';
  const notesLabel=labels.notes?.previousElementSibling;
  if(notesLabel)notesLabel.textContent='PRISE DE NOTES — automatique depuis les exercices';
  const summaryLabel=labels.summary?.previousElementSibling;
  if(summaryLabel)summaryLabel.textContent='SYNTHÈSE — automatique avec ton propre vocabulaire';

  const controls=document.createElement('div');
  controls.className='actions';
  controls.style.padding='10px 14px 0';
  controls.innerHTML='<button class="btn" id="refreshCornellV38" type="button">Actualiser depuis mes exercices</button>';
  sheet.insertBefore(controls,sheet.firstChild);

  const coreDefs=[
    ['What?','what'],['Who?','who'],['Where?','where'],['When?','when'],['Why?','why'],['How?','how'],
    ['What if?','whatif'],['Feynman','feynman'],['Exemple concret','example'],['Analogie','analogy'],
    ['Exercices pratiques / rappel actif','practice'],['Méthode des erreurs','errors'],['Mini-examen','miniTest']
  ];

  const STOP=new Set(('le la les un une des du de d et ou mais donc car que qui quoi quand comment pourquoi où au aux en dans sur sous avec sans pour par ce cet cette ces mon ma mes ton ta tes son sa ses notre nos votre vos leur leurs je tu il elle nous vous ils elles se ne pas plus moins très trop aussi encore déjà puis alors comme est sont suis es être avoir a ont ai as fait faire cela ça ceci celle celui ceux on y à l the a an and or but of to in on for with from is are was were be been being this that these those i you he she we they my your our their').split(/s+/));

  function words(text){
    return clean(text)
      .toLocaleLowerCase('fr')
      .normalize('NFD').replace(/[̀-ͯ]/g,'')
      .replace(/[^a-z0-9+#.-]+/g,' ')
      .split(/s+/)
      .filter(w=>w.length>=3&&!STOP.has(w)&&!/^d+$/.test(w));
  }

  function extractKeywords(text,topic){
    const counts=new Map();
    const add=(w,weight=1)=>counts.set(w,(counts.get(w)||0)+weight);
    words(text).forEach(w=>add(w,1));
    words(topic).forEach(w=>add(w,4));
    return [...counts.entries()]
      .sort((a,b)=>b[1]-a[1]||a[0].localeCompare(b[0]))
      .slice(0,16)
      .map(([w])=>w)
      .join(' · ');
  }

  function uniquePush(arr,value){
    const v=clean(value);
    if(!v)return;
    const key=v.toLocaleLowerCase('fr');
    if(!arr.some(x=>x.toLocaleLowerCase('fr')===key))arr.push(v);
  }

  function collectSource(){
    const learn=readJson(LEARN_KEY,{})||{};
    const work=readJson(WORK_KEY,{})||{};

    const sections=[];
    const rawAnswers=[];
    const coreLines=[];

    coreDefs.forEach(([label,key])=>{
      const v=clean(learn[key]);
      if(!v)return;
      coreLines.push(label+' : '+v);
      rawAnswers.push(v);
    });
    if(coreLines.length)sections.push('CYCLE PRINCIPAL\n'+coreLines.join('\n'));

    METHOD_DEFS.forEach(([title,fields],index)=>{
      const saved=work[String(index)]||{};
      const answers=Array.isArray(saved.answers)?saved.answers:[];
      const lines=[];
      answers.forEach((value,i)=>{
        const v=clean(value);
        if(!v)return;
        lines.push((fields[i]||('Question '+(i+1)))+'\n→ '+v);
        rawAnswers.push(v);
      });
      if(lines.length){
        const topic=clean(saved.topic);
        sections.push(title+(topic?' — '+topic:'')+'\n'+lines.join('\n'));
      }
    });

    return{
      learn,
      topic:clean(learn.topic),
      notes:sections.join('\n\n'),
      rawAnswers
    };
  }

  function makeSummary(source){
    const picks=[];
    const priority=['feynman','what','why','how','example','whatif'];
    priority.forEach(k=>uniquePush(picks,source.learn[k]));

    if(picks.length<5){
      source.rawAnswers.forEach(v=>{
        if(picks.length>=5)return;
        uniquePush(picks,v);
      });
    }

    const chosen=picks.slice(0,5);
    if(!chosen.length)return '';

    let out=source.topic?'Sujet : '+source.topic+'.\n\n':'';
    out+='Je retiens : '+chosen.map(v=>{
      const t=v.replace(/s+/g,' ').trim();
      return /[.!?]$/.test(t)?t:t+'.';
    }).join(' ');
    return out;
  }

  function updateCornell(){
    // Sauvegarder d'abord ce qui est actuellement saisi au-dessus.
    const modal=document.getElementById('methodModalV35');
    if(modal&&modal.classList.contains('open'))document.getElementById('methodSaveV35')?.click();
    document.getElementById('learningSave')?.click();

    const source=collectSource();
    const current=readJson(CORNELL_KEY,{})||{};

    // V47: once a complete V45/V47 verification has populated Cornell,
    // do not replace the corrected material with the older raw exercise digest.
    const verified=readJson('second-brain-verify-v43',null);
    if(current.verifiedByV45===true && verified?.schema_version===45 && verified?.verification_complete===true){
      const ids={title:'cornellTitleV37',keywords:'cornellKeywordsV37',formulas:'cornellFormulasV37',doubts:'cornellDoubtsV37',essentials:'cornellEssentialsV37',dates:'cornellDatesV37',authors:'cornellAuthorsV37',notes:'cornellNotesV37',summary:'cornellSummaryV37'};
      Object.entries(ids).forEach(([k,id])=>{const el=document.getElementById(id);if(el&&current[k]!==undefined)el.value=current[k]||'';});
      const st=document.getElementById('cornellStatusV37');
      if(st)st.textContent='Cornell conservée depuis la dernière correction vérifiée. Modifie une réponse pour revenir au mode exercices.';
      return;
    }

    const generated={
      title:source.topic||'',
      keywords:extractKeywords(source.rawAnswers.join(' '),source.topic),
      notes:source.notes,
      summary:makeSummary(source)
    };

    const merged={
      ...current,
      ...generated,
      generatedFromExercises:true,
      updatedAt:new Date().toISOString()
    };

    try{
      localStorage.setItem(CORNELL_KEY,JSON.stringify(merged));
      if(labels.title)labels.title.value=generated.title;
      if(labels.keywords)labels.keywords.value=generated.keywords;
      if(labels.notes)labels.notes.value=generated.notes;
      if(labels.summary)labels.summary.value=generated.summary;
      const st=document.getElementById('cornellStatusV37');
      if(st)st.textContent='Cornell actualisée automatiquement depuis tes exercices : '+new Date().toLocaleTimeString('fr-CA');
    }catch(e){
      const st=document.getElementById('cornellStatusV37');
      if(st)st.textContent='Impossible d’actualiser Cornell : stockage du navigateur plein.';
    }
  }

  document.getElementById('refreshCornellV38').onclick=updateCornell;

  let timer=null;
  function queueRefresh(){
    clearTimeout(timer);
    timer=setTimeout(updateCornell,550);
  }

  page.addEventListener('input',e=>{
    if(e.target.closest('#cornellSheetV37'))return;
    queueRefresh();
  });
  page.addEventListener('change',e=>{
    if(e.target.closest('#cornellSheetV37'))return;
    queueRefresh();
  });

  const modal=document.getElementById('methodModalV35');
  if(modal){
    modal.addEventListener('input',queueRefresh);
    modal.addEventListener('change',queueRefresh);
  }

  const originalSave=document.getElementById('learningSave');
  if(originalSave)originalSave.addEventListener('click',()=>setTimeout(updateCornell,50));

  const methodSave=document.getElementById('methodSaveV35');
  if(methodSave)methodSave.addEventListener('click',()=>setTimeout(updateCornell,50));

  // S'assurer que l'export Word relit une Cornell fraîche.
  ['learningExportWordV36','methodExportWordV36'].forEach(id=>{
    const btn=document.getElementById(id);
    if(!btn||btn.dataset.v38Wrapped)return;
    btn.dataset.v38Wrapped='1';
    btn.addEventListener('click',updateCornell,true);
  });

  updateCornell();
})();
