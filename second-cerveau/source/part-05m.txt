
// V26 — ajout du nutriment sur la foi et les buts
(()=>{
  try{
    document.title='Second Cerveau IA — V26';
    const pill=document.querySelector('.pill');if(pill)pill.textContent='Second Cerveau IA — V26';
    const vs=document.getElementById('versionStatus');if(vs)vs.textContent='Version actuelle V26 · cette version hérite directement de V25 et ajoute dans Nutriments la question sur la foi et l’atteinte des buts.';
    const versions=document.getElementById('versionList');
    if(versions&&!versions.dataset.v26){
      versions.dataset.v26='1';
      versions.insertAdjacentHTML('afterbegin','<div class="item"><strong>V26 · actuelle</strong><small>Ajout dans Ma relation avec Jésus → Nutriments : « Est-ce que ma foi me permet d’atteindre mes buts ? », sans supprimer les réponses existantes.</small></div>');
      const old=[...versions.querySelectorAll('strong')].find(x=>x.textContent.includes('V25 · actuelle'));if(old)old.textContent='V25';
    }
  }catch(e){}
})();
