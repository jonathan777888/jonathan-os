    status('Ouverture de Google…');
    const result=await client.auth.signInWithOAuth({provider:'google',options:{redirectTo:REDIRECT_URL}});
    if(result.error)status('Connexion Google impossible : '+result.error.message,'warn');
  }
  async function signOut(){
    if(!client)return;
    await client.auth.signOut();
    user=null;bootedUserId=null;pendingRemote=null;hideConflict();renderAuth();
  }
  async function fetchRemote(){
    if(!client||!user)return null;
    const result=await client.from(TABLE).select('payload,updated_at,device_label').eq('user_id',user.id).maybeSingle();
    if(result.error)throw result.error;
    return result.data||null;
  }
  async function pushNow(message){
    if(!client||!user||syncing||pendingRemote)return;
    clearTimeout(pushTimer);
    syncing=true;status('Envoi des données vers le cloud…');
    try{
      const row={user_id:user.id,payload:snapshot(),device_label:(navigator.platform||'navigateur')+' · '+(navigator.userAgent||'').slice(0,120)};
      const result=await client.from(TABLE).upsert(row,{onConflict:'user_id'}).select('updated_at').single();
      if(result.error)throw result.error;
      const now=new Date().toISOString();
      writeMeta({userId:user.id,initialized:true,lastRemoteUpdatedAt:result.data.updated_at,lastSyncedAt:now});
      status(message||'Données synchronisées avec le cloud.','ok');
    }catch(e){
      status('Synchronisation impossible : '+(e&&e.message?e.message:String(e)),'warn');
    }finally{syncing=false}
  }
  async function applyRemote(remote){
    if(!remote||!remote.payload||!user)return;
    applying=true;
    try{
      const payload=remote.payload||{};
      const localKeys=[];
      for(let i=0;i<localStorage.length;i++){const k=localStorage.key(i);if(isSyncableKey(k))localKeys.push(k)}
      localKeys.forEach(k=>{if(!Object.prototype.hasOwnProperty.call(payload,k))nativeRemove.call(localStorage,k)});
      Object.entries(payload).forEach(([k,v])=>{if(isSyncableKey(k)&&typeof v==='string')nativeSet.call(localStorage,k,v)});
      const now=new Date().toISOString();
      writeMeta({userId:user.id,initialized:true,lastRemoteUpdatedAt:remote.updated_at,lastSyncedAt:now,lastLocalChangeAt:null});
      pendingRemote=null;hideConflict();status('Données du cloud récupérées. Rechargement…','ok');
      setTimeout(()=>location.reload(),250);
    }catch(e){
      status('Impossible de restaurer les données du cloud : '+(e&&e.message?e.message:String(e)),'warn');
    }finally{applying=false}
  }
  async function initialSync(){
    if(!user)return;
    try{
      const remote=await fetchRemote();
      const meta=readMeta();
      if(!remote){await pushNow('Première sauvegarde cloud créée.');return}
      if(meta.userId!==user.id||!meta.initialized){
        showConflict(remote,'Ce compte possède déjà des données dans le cloud. Choisis la version à conserver avant la première synchronisation sur cet appareil.');
        return;
      }
      if(time(remote.updated_at)>time(meta.lastRemoteUpdatedAt)+500){
        if(localDirty(meta))showConflict(remote,'Le cloud et cet appareil ont tous les deux été modifiés.');
        else await applyRemote(remote);
        return;
      }
      if(localDirty(meta))await pushNow();
      else status('Tout est à jour.','ok');
    }catch(e){
      status('Vérification du cloud impossible : '+(e&&e.message?e.message:String(e)),'warn');
    }
  }
  async function manualSync(){
    if(!user){await signIn();return}
    if(pendingRemote){status('Choisis d’abord quelle version tu veux conserver.','warn');return}
    try{
      const remote=await fetchRemote(),meta=readMeta();
      if(!remote){await pushNow();return}
      if(time(remote.updated_at)>time(meta.lastRemoteUpdatedAt)+500){
        if(localDirty(meta))showConflict(remote,'Le cloud et cet appareil ont tous les deux été modifiés.');
        else await applyRemote(remote);
      }else await pushNow();
    }catch(e){status('Synchronisation impossible : '+(e&&e.message?e.message:String(e)),'warn')}
  }
  async function checkRemote(){
    if(!user||syncing||pendingRemote||document.hidden)return;
    try{
      const remote=await fetchRemote();
      if(!remote)return;
      const meta=readMeta();
      if(time(remote.updated_at)>time(meta.lastRemoteUpdatedAt)+500){
        if(localDirty(meta))showConflict(remote,'De nouvelles données existent dans le cloud et cet appareil contient aussi des changements.');
        else await applyRemote(remote);
      }
    }catch(e){}
  }
  function markDirty(){
    writeMeta({userId:user?user.id:readMeta().userId||null,lastLocalChangeAt:new Date().toISOString()});
    if(user&&!pendingRemote){
      status('Modification locale détectée. Synchronisation automatique en attente…');
      clearTimeout(pushTimer);pushTimer=setTimeout(()=>pushNow(),1400);
    }
  }

  Storage.prototype.setItem=function(key,value){
    const result=nativeSet.call(this,key,value);
    if(this===localStorage&&isSyncableKey(key)&&!applying)markDirty();
    return result;
  };
  Storage.prototype.removeItem=function(key){
    const result=nativeRemove.call(this,key);
    if(this===localStorage&&isSyncableKey(key)&&!applying)markDirty();
    return result;
  };

  window.SecondBrainCloudV52={snapshot,isSyncableKey,manualSync};

  async function start(){
    installUI();
    try{
      await loadLibrary();
      client=window.supabase.createClient(SUPABASE_URL,SUPABASE_KEY,{auth:{persistSession:true,autoRefreshToken:true,detectSessionInUrl:true}});
      const sessionResult=await client.auth.getSession();
      user=sessionResult.data&&sessionResult.data.session?sessionResult.data.session.user:null;
      renderAuth();
      if(user){bootedUserId=user.id;await initialSync()}
      client.auth.onAuthStateChange((event,session)=>{
        const next=session&&session.user?session.user:null;
        user=next;renderAuth();
        if(user&&bootedUserId!==user.id){bootedUserId=user.id;setTimeout(()=>initialSync(),0)}
        if(!user)bootedUserId=null;
      });
      remoteTimer=setInterval(checkRemote,60000);
      document.addEventListener('visibilitychange',()=>{if(!document.hidden)checkRemote()});
    }catch(e){
      status('Le module de synchronisation n’a pas pu démarrer. Tes données locales restent disponibles.','warn');
    }
  }
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',start,{once:true});else start();
})();
