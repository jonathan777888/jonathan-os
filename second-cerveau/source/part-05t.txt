      <div class="card section-card">

            <p>Feynman · Rappel actif · Répétition espacée · Exercices pratiques · Pourquoi/Comment · Cartes mentales · Double codage · Exemples concrets · Comparaison · Apprentissage par projet · Entrelacement · Auto-évaluation · Cornell · SQ3R · Enseigner à quelqu’un · Méthode des erreurs · Analogies · Chunking · Pré-test · Révision cumulative.</p>

    const backupPage=document.getElementById('page-backup');
    if(backupPage)backupPage.parentElement.insertBefore(page,backupPage);else document.querySelector('.app').appendChild(page);
  }

  const ids={
    topic:'learnTopic',what:'learnWhat',who:'learnWho',where:'learnWhere',when:'learnWhen',why:'learnWhy',how:'learnHow',whatif:'learnWhatIf',
    feynman:'learnFeynman',example:'learnExample',analogy:'learnAnalogy',practice:'learnPractice',errors:'learnErrors',miniTest:'learnMiniTest'
  };
  const checks={reviewJ1:'learnJ1',reviewJ3:'learnJ3',reviewJ7:'learnJ7',reviewJ14:'learnJ14',reviewJ30:'learnJ30'};

  function read(){
    try{return Object.assign({},DEFAULTS,JSON.parse(localStorage.getItem(STORAGE_KEY)||'null')||{})}
    catch(e){return Object.assign({},DEFAULTS)}
  }
  function collect(){
    const s={};
    Object.entries(ids).forEach(([k,id])=>s[k]=document.getElementById(id)?.value||'');
    Object.entries(checks).forEach(([k,id])=>s[k]=!!document.getElementById(id)?.checked);
    s.updatedAt=new Date().toISOString();
    return s;
  }
  function saveLearning(message='Autosauvegardé.'){
    try{
      localStorage.setItem(STORAGE_KEY,JSON.stringify(collect()));
      const st=document.getElementById('learningStatus');if(st)st.textContent=message;
    }catch(e){
      const st=document.getElementById('learningStatus');if(st)st.textContent='Impossible de sauvegarder : stockage du navigateur plein. Exporte une sauvegarde du Second Cerveau.';
    }
  }
  function load(){
    const s=read();
    Object.entries(ids).forEach(([k,id])=>{const el=document.getElementById(id);if(el)el.value=s[k]||''});
    Object.entries(checks).forEach(([k,id])=>{const el=document.getElementById(id);if(el)el.checked=!!s[k]});
    const st=document.getElementById('learningStatus');
    if(st&&s.updatedAt)st.textContent='Dernière sauvegarde : '+new Date(s.updatedAt).toLocaleString('fr-CA');
  }

  let timer=null;
  [...Object.values(ids),...Object.values(checks)].forEach(id=>{
    const el=document.getElementById(id);if(!el)return;
    const event=el.type==='checkbox'?'change':'input';
    el.addEventListener(event,()=>{
      clearTimeout(timer);
      timer=setTimeout(()=>saveLearning(),220);
    });
  });

  document.getElementById('learningSave').onclick=()=>saveLearning('Apprentissage enregistré.');
  document.getElementById('learningNew').onclick=()=>{
    if(!confirm('Commencer un nouveau sujet ? Le sujet actuel sera effacé de cet onglet.'))return;
    try{localStorage.removeItem(STORAGE_KEY)}catch(e){}
    Object.values(ids).forEach(id=>{const el=document.getElementById(id);if(el)el.value=''});
    Object.values(checks).forEach(id=>{const el=document.getElementById(id);if(el)el.checked=false});
    const st=document.getElementById('learningStatus');if(st)st.textContent='Nouveau sujet prêt.';
  };

  tab.onclick=()=>{
    document.querySelectorAll('.tab').forEach(x=>x.classList.toggle('active',x===tab));
    document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
    page.classList.add('active');
    window.scrollTo({top:0,behavior:'smooth'});
  };

  load();
})();
